#include <Keypad.h>
#include <Servo.h>

Servo myServo;

// === Password Settings ===
String password = "1234";  // Set your password here
String inputPassword = "";

// === Keypad Configuration ===
const byte ROWS = 4; // Four rows
const byte COLS = 4; // Four columns
char keys[ROWS][COLS] = {
  {'1','2','3','A'},
  {'4','5','6','B'},
  {'7','8','9','C'},
  {'*','0','#','D'}
};
byte rowPins[ROWS] = {9, 8, 7, 6}; // Connect to R1, R2, R3, R4
byte colPins[COLS] = {5, 4, 3, 2}; // Connect to C1, C2, C3, C4

Keypad keypad = Keypad(makeKeymap(keys), rowPins, colPins, ROWS, COLS);

void setup() {
  myServo.attach(10);       // Servo pin
  myServo.write(0);         // Start in LOCK position
  Serial.begin(9600);
  Serial.println("Enter Password:");
}

void loop() {
  char key = keypad.getKey();

  if (key) {
    if (key == '#') { // '#' to confirm entry
      if (inputPassword == password) {
        Serial.println("Access Granted ✅");
        unlockDoor();
      } else {
        Serial.println("Access Denied ❌");
        lockDoor();
      }
      inputPassword = ""; // Reset input
    }
    else if (key == '') { // '' to clear
      inputPassword = "";
      Serial.println("Input Cleared");
    }
    else {
      inputPassword += key; // Append key to input
      Serial.print("*");    // Hide actual input
    }
  }
}

void unlockDoor() {
  myServo.write(90); // Rotate to UNLOCK
  delay(3000);       // Keep door unlocked for 3 seconds
  lockDoor();
}

void lockDoor() {
  myServo.write(0); // Rotate back to LOCK
}
